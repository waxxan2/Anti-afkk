-- Sürükleme (Draggable) Özelliği
local dragging, dragInput, dragStart, startPos
MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

game:GetService("UserInputService").InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

-- FPS ve MS Hesaplama Döngüsü
local frameCount = 0
local lastUpdate = tick()
RunService.RenderStepped:Connect(function()
    frameCount = frameCount + 1
    local now = tick()
    if now - lastUpdate >= 1 then
        local fps = math.floor(frameCount / (now - lastUpdate))
        FpsLabel.Text = "FPS: " .. fps
        
        pcall(function()
            local ping = math.floor(Stats.Network.ServerStatsItem["Data Ping"]:GetValue())
            MsLabel.Text = "MS (Ping): " .. ping .. " ms"
        end)
        
        frameCount = 0
        lastUpdate = now
    end
end)

-- Oyunda Kalınan Süre Hesaplama
local sessionStart = tick()
task.spawn(function()
    while task.wait(1) do
        local totalSeconds = math.floor(tick() - sessionStart)
        local hours = math.floor(totalSeconds / 3600)
        local minutes = math.floor((totalSeconds % 3600) / 60)
        local seconds = totalSeconds % 60
        TimeLabel.Text = string.format("Süre: %02d:%02d:%02d", hours, minutes, seconds)
    end
end)

-- Anti-AFK Mantığı ve Kütüphane Entegrasyonu (Başlangıçta Kapalı)
local antiAfkThread = nil
local uiState = false

local function ToggleAntiAfk(Value)
    uiState = Value
    if Value then
        ToggleButton.Text = "Anti-AFK: AÇIK"
        TweenService:Create(ToggleButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(180, 20, 40)}):Play()
        TweenService:Create(ToggleButton, TweenInfo.new(0.2), {TextColor3 = Color3.fromRGB(255, 255, 255)}):Play()
        
        antiAfkThread = task.spawn(function()
            while uiState do
                task.wait(600) -- 10 Dakika
                if not uiState then break end
                pcall(function()
                    VirtualUser:Button2Down(Vector2.new(0, 0), workspace.CurrentCamera.CFrame)
                    task.wait(1)
                    VirtualUser:Button2Up(Vector2.new(0, 0), workspace.CurrentCamera.CFrame)
                end)
            end
        end)
    else
        ToggleButton.Text = "Anti-AFK: KAPALI"
        TweenService:Create(ToggleButton, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(40, 40, 40)}):Play()
        TweenService:Create(ToggleButton, TweenInfo.new(0.2), {TextColor3 = Color3.fromRGB(170, 170, 170)}):Play()
        
        if antiAfkThread then
            task.cancel(antiAfkThread)
            antiAfkThread = nil
        end
    end
end

-- Mini UI üzerindeki butona basılınca tetiklenme
ToggleButton.MouseButton1Click:Connect(function()
    ToggleAntiAfk(not uiState)
end)

-- Kütüphane Fonksiyonu (Başlangıçta false gelir, çalışmaz)
Misc:AddSwatch("⏰ Anti Afk", function(Value)
    ToggleAntiAfk(Value)
end)
